cmake_minimum_required(VERSION 3.20)
project(ContextTests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(context_tests ContextTests.cpp ../lab1/CryptoContext.cpp)
target_compile_features(context_tests PRIVATE cxx_std_20)
target_link_libraries(context_tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(context_tests PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(context_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME context_tests COMMAND context_tests ${CMAKE_CURRENT_BINARY_DIR})
